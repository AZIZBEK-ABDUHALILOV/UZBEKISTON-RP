cmake_minimum_required(VERSION 3.22)
project(samp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB_RECURSE SRC ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_library(samp SHARED ${SRC})
target_include_directories(samp PRIVATE src)
target_compile_options(samp PRIVATE -fvisibility=hidden -Wall)
target_link_libraries(samp log android dl)
